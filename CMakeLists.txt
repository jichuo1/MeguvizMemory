cmake_minimum_required(VERSION 3.25)
project(MeguvizMemory VERSION 0.2.0 LANGUAGES C CXX)

option(BUILD_SHARED_LIBS "Build the public shared library" ON)
option(MGMEM_BUILD_TESTS "Build ownership and failure-path verification" ON)
option(MGMEM_BUILD_JNI_HOST_TESTS "Build header-only fake-VM JNI lock regressions on host" OFF)
option(MGMEM_BUILD_BENCHMARKS "Build bounded owned-region diagnostic" ON)
option(MGMEM_SANITIZE "Enable address and undefined-behavior diagnostics" OFF)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
add_library(mgmem src/mgmem.cpp)
add_library(MeguvizMemory::mgmem ALIAS mgmem)
target_compile_features(mgmem PUBLIC cxx_std_17)
target_include_directories(mgmem PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(mgmem PRIVATE Threads::Threads)
add_library(mgmem_pressure STATIC adapters/android/native/pressure.cpp)
add_library(MeguvizMemory::pressure ALIAS mgmem_pressure)
target_compile_features(mgmem_pressure PUBLIC cxx_std_17)
target_include_directories(mgmem_pressure PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(mgmem_pressure PUBLIC mgmem Threads::Threads)
set_target_properties(mgmem_pressure PROPERTIES EXPORT_NAME pressure POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
set_target_properties(mgmem PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON VERSION ${PROJECT_VERSION} SOVERSION 1)
if(BUILD_SHARED_LIBS)
  target_compile_definitions(mgmem PUBLIC MGMEM_SHARED PRIVATE MGMEM_BUILDING)
endif()
if(MSVC)
  target_compile_options(mgmem PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(mgmem_pressure PRIVATE /W4 /permissive- /utf-8)
  if(MGMEM_SANITIZE)
    target_compile_options(mgmem PRIVATE /fsanitize=address /Zi)
    target_link_options(mgmem PRIVATE /INCREMENTAL:NO)
    target_compile_options(mgmem_pressure PRIVATE /fsanitize=address /Zi)
  endif()
else()
  target_compile_options(mgmem PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(mgmem_pressure PRIVATE -Wall -Wextra -Wpedantic)
  if(MGMEM_SANITIZE)
    target_compile_options(mgmem PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mgmem PUBLIC -fsanitize=address,undefined)
    target_compile_options(mgmem_pressure PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
endif()
if(ANDROID)
  target_link_options(mgmem PRIVATE "-Wl,-z,max-page-size=16384")
  install(FILES third_party_notices/llvm-NOTICE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizMemory/licenses)
endif()

install(TARGETS mgmem EXPORT MeguvizMemoryTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(TARGETS mgmem_pressure EXPORT MeguvizMemoryTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT MeguvizMemoryTargets NAMESPACE MeguvizMemory:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizMemory)
configure_package_config_file(cmake/MeguvizMemoryConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/MeguvizMemoryConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizMemory)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MeguvizMemoryConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MeguvizMemoryConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/MeguvizMemoryConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizMemory)

if(MGMEM_BUILD_TESTS)
  enable_testing()
  add_executable(mgmem_core_tests tests/core_tests.cpp)
  target_link_libraries(mgmem_core_tests PRIVATE mgmem Threads::Threads)
  target_compile_features(mgmem_core_tests PRIVATE cxx_std_17)
  add_test(NAME ownership_budget_close COMMAND mgmem_core_tests)
  set_tests_properties(ownership_budget_close PROPERTIES TIMEOUT 60)
  add_executable(mgmem_c_consumer tests/c_consumer.c)
  target_link_libraries(mgmem_c_consumer PRIVATE mgmem)
  add_test(NAME c_abi_consumer COMMAND mgmem_c_consumer)
  add_executable(mgmem_cpp_consumer tests/cpp_consumer.cpp)
  target_link_libraries(mgmem_cpp_consumer PRIVATE mgmem Threads::Threads)
  target_compile_features(mgmem_cpp_consumer PRIVATE cxx_std_17)
  add_test(NAME cpp_raii_consumer COMMAND mgmem_cpp_consumer)
  add_executable(mgmem_import_tests tests/import_tests.cpp)
  target_compile_features(mgmem_import_tests PRIVATE cxx_std_17)
  target_link_libraries(mgmem_import_tests PRIVATE mgmem Threads::Threads)
  add_test(NAME external_import COMMAND mgmem_import_tests)
  foreach(mode retain release alias)
    add_test(NAME external_import_poison_${mode} COMMAND mgmem_import_tests --poison-${mode})
    set_tests_properties(external_import_poison_${mode} PROPERTIES TIMEOUT 20)
  endforeach()
  set_tests_properties(external_import PROPERTIES TIMEOUT 60)
  add_executable(mgmem_pressure_tests tests/pressure_tests.cpp)
  target_compile_features(mgmem_pressure_tests PRIVATE cxx_std_17)
  target_link_libraries(mgmem_pressure_tests PRIVATE mgmem_pressure)
  add_test(NAME pressure_budget_close COMMAND mgmem_pressure_tests)
  if(MGMEM_BUILD_JNI_HOST_TESTS)
    if(ANDROID)
      message(FATAL_ERROR "MGMEM_BUILD_JNI_HOST_TESTS uses host JDK headers only")
    endif()
    find_package(JNI REQUIRED)
    add_executable(mgmem_jni_lock_tests tests/jni_lock_tests.cpp)
    target_compile_features(mgmem_jni_lock_tests PRIVATE cxx_std_17)
    target_include_directories(mgmem_jni_lock_tests PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(mgmem_jni_lock_tests PRIVATE mgmem_pressure Threads::Threads)
    # JNI headers/function tables only: do not load or link a JVM for these tests.
    if(MSVC)
      target_compile_options(mgmem_jni_lock_tests PRIVATE /W4 /permissive- /utf-8)
      if(MGMEM_SANITIZE)
        target_compile_options(mgmem_jni_lock_tests PRIVATE /fsanitize=address /Zi)
        target_link_options(mgmem_jni_lock_tests PRIVATE /INCREMENTAL:NO)
      endif()
    else()
      target_compile_options(mgmem_jni_lock_tests PRIVATE -Wall -Wextra -Wpedantic)
      if(MGMEM_SANITIZE)
        target_compile_options(mgmem_jni_lock_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(mgmem_jni_lock_tests PRIVATE -fsanitize=address,undefined)
      endif()
    endif()
    foreach(case callback-reentry single-publisher attach-retry java-exception-retry error-quarantine foreign-release foreign-release-bind foreign-release-snapshot auto-attach-recovery auto-java-recovery auto-concurrent-poll auto-error-quarantine binding-exception-matrix snapshot-error-external snapshot-error-owned snapshot-state-external snapshot-state-owned snapshot-header-external snapshot-header-owned)
      add_test(NAME jni_lock_${case} COMMAND mgmem_jni_lock_tests ${case})
      set_tests_properties(jni_lock_${case} PROPERTIES TIMEOUT 10)
    endforeach()
  endif()
  set_tests_properties(pressure_budget_close PROPERTIES TIMEOUT 60)
  add_executable(mgmem_abi10_consumer tests/compat/abi10_consumer.c)
  target_include_directories(mgmem_abi10_consumer BEFORE PRIVATE tests/compat/abi10/include)
  target_link_libraries(mgmem_abi10_consumer PRIVATE mgmem)
  add_test(NAME original_abi10_consumer COMMAND mgmem_abi10_consumer)
  set_tests_properties(original_abi10_consumer PROPERTIES TIMEOUT 30)
  if(BUILD_SHARED_LIBS AND NOT ANDROID)
    add_library(mgmem_other_runtime SHARED src/mgmem.cpp)
    target_include_directories(mgmem_other_runtime PRIVATE include)
    target_compile_features(mgmem_other_runtime PRIVATE cxx_std_17)
    target_compile_definitions(mgmem_other_runtime PRIVATE MGMEM_SHARED MGMEM_BUILDING)
    target_link_libraries(mgmem_other_runtime PRIVATE Threads::Threads)
    add_executable(mgmem_runtime_tests tests/runtime_tests.cpp)
    target_include_directories(mgmem_runtime_tests PRIVATE include)
    target_compile_features(mgmem_runtime_tests PRIVATE cxx_std_17)
    target_link_libraries(mgmem_runtime_tests PRIVATE ${CMAKE_DL_LIBS})
    add_test(NAME runtime_origin COMMAND mgmem_runtime_tests $<TARGET_FILE:mgmem> $<TARGET_FILE:mgmem_other_runtime>)
    set_tests_properties(runtime_origin PROPERTIES TIMEOUT 30)
  endif()
  set_tests_properties(c_abi_consumer cpp_raii_consumer PROPERTIES TIMEOUT 30)
  if(MSVC AND MGMEM_SANITIZE)
    foreach(target mgmem_core_tests mgmem_c_consumer mgmem_cpp_consumer mgmem_import_tests mgmem_pressure_tests mgmem_abi10_consumer)
      target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
      target_link_options(${target} PRIVATE /INCREMENTAL:NO)
    endforeach()
  elseif(MGMEM_SANITIZE)
    foreach(target mgmem_core_tests mgmem_c_consumer mgmem_cpp_consumer mgmem_import_tests mgmem_pressure_tests mgmem_abi10_consumer)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endforeach()
  endif()
  if(MGMEM_SANITIZE AND TARGET mgmem_other_runtime)
    if(MSVC)
      target_compile_options(mgmem_other_runtime PRIVATE /fsanitize=address /Zi)
      target_link_options(mgmem_other_runtime PRIVATE /INCREMENTAL:NO)
    else()
      target_compile_options(mgmem_other_runtime PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(mgmem_other_runtime PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endif()
if(MGMEM_BUILD_BENCHMARKS)
  add_executable(mgmem_tlb_owned bench/tlb_owned.cpp)
  target_compile_features(mgmem_tlb_owned PRIVATE cxx_std_17)
  target_link_libraries(mgmem_tlb_owned PRIVATE mgmem)
  install(TARGETS mgmem_tlb_owned RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()

set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "MeguvizMemory-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
