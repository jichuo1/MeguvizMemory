cmake_minimum_required(VERSION 3.25)
project(MeguvizMemoryJavaImageConsumer LANGUAGES CXX)
find_package(MeguvizMemory CONFIG REQUIRED)
find_package(JNI REQUIRED)
find_package(Java REQUIRED COMPONENTS Runtime Development)
find_package(Threads REQUIRED)
get_target_property(memory_kind MeguvizMemory::mgmem TYPE)
if(NOT memory_kind STREQUAL "SHARED_LIBRARY")
  message(FATAL_ERROR "The consumer requires the installed shared Memory runtime")
endif()
set(MGMEM_JNI_PREFIX "" CACHE PATH "Installed host JNI runtime prefix")
set(MGMEM_ANDROID_CLASSES_JAR "" CACHE FILEPATH "classes.jar extracted from the matching Android AAR")
if(NOT IS_DIRECTORY "${MGMEM_JNI_PREFIX}" OR NOT EXISTS "${MGMEM_ANDROID_CLASSES_JAR}")
  message(FATAL_ERROR "Set MGMEM_JNI_PREFIX and MGMEM_ANDROID_CLASSES_JAR to installed inputs")
endif()
find_library(MGMEM_JNI_LINK_LIBRARY NAMES mgmem_pressure_jni PATHS "${MGMEM_JNI_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
if(WIN32)
  find_file(MGMEM_JNI_RUNTIME_LIBRARY NAMES mgmem_pressure_jni.dll PATHS "${MGMEM_JNI_PREFIX}/bin" NO_DEFAULT_PATH REQUIRED)
  add_library(mgmem_installed_jni SHARED IMPORTED)
  set_target_properties(mgmem_installed_jni PROPERTIES IMPORTED_LOCATION "${MGMEM_JNI_RUNTIME_LIBRARY}" IMPORTED_IMPLIB "${MGMEM_JNI_LINK_LIBRARY}")
else()
  add_library(mgmem_installed_jni SHARED IMPORTED)
  set_target_properties(mgmem_installed_jni PROPERTIES IMPORTED_LOCATION "${MGMEM_JNI_LINK_LIBRARY}")
endif()
add_library(mgmem_java_image SHARED native/image_jni.cpp)
target_compile_features(mgmem_java_image PRIVATE cxx_std_17)
target_include_directories(mgmem_java_image PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(mgmem_java_image PRIVATE MeguvizMemory::mgmem mgmem_installed_jni Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_compile_definitions(mgmem_java_image PRIVATE MGMEM_PRESSURE_SHARED)
endif()
if(MSVC)
  target_compile_options(mgmem_java_image PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(mgmem_java_image PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(mgmem_java_image PROPERTIES CXX_EXTENSIONS OFF)
set(JAVA_CLASSES "${CMAKE_CURRENT_BINARY_DIR}/classes")
add_custom_command(OUTPUT "${JAVA_CLASSES}/io/github/jichuo1/meguviz/memory/image/JavaImage.class"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${JAVA_CLASSES}"
  COMMAND ${Java_JAVAC_EXECUTABLE} --release 8 -classpath "${MGMEM_ANDROID_CLASSES_JAR}" -d "${JAVA_CLASSES}"
    "${CMAKE_CURRENT_SOURCE_DIR}/java/io/github/jichuo1/meguviz/memory/image/JavaImage.java"
  DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/java/io/github/jichuo1/meguviz/memory/image/JavaImage.java" "${MGMEM_ANDROID_CLASSES_JAR}"
  VERBATIM)
add_custom_target(java_image_classes ALL DEPENDS "${JAVA_CLASSES}/io/github/jichuo1/meguviz/memory/image/JavaImage.class")
