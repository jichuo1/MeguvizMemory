cmake_minimum_required(VERSION 3.25)
project(MeguvizMemoryImageConsumer LANGUAGES CXX)
find_package(MeguvizMemory CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(mgmem_image main.cpp)
add_executable(mgmem_image_paired paired.cpp)
target_compile_features(mgmem_image PRIVATE cxx_std_17)
target_link_libraries(mgmem_image PRIVATE MeguvizMemory::mgmem Threads::Threads)
target_compile_features(mgmem_image_paired PRIVATE cxx_std_17)
target_link_libraries(mgmem_image_paired PRIVATE MeguvizMemory::mgmem Threads::Threads)
if(MSVC)
  target_compile_options(mgmem_image PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(mgmem_image_paired PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(mgmem_image PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(mgmem_image_paired PRIVATE -Wall -Wextra -Wpedantic)
endif()
