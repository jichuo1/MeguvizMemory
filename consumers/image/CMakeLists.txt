cmake_minimum_required(VERSION 3.25)
project(MeguvizMemoryImageConsumer LANGUAGES CXX)
find_package(MeguvizMemory CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(mgmem_image main.cpp)
add_executable(mgmem_image_paired paired.cpp)
add_executable(mgmem_image_perf perf.cpp)
add_executable(mgmem_image_pressure_stability pressure_stability.cpp)
option(MGMEM_IMAGE_BUILD_TRACE "Build explicit image trace capture diagnostics" OFF)
option(MGMEM_IMAGE_BUILD_TRACE_TESTS "Build trace fault-injection diagnostic" OFF)
if(MGMEM_IMAGE_BUILD_TRACE_TESTS AND NOT MGMEM_IMAGE_BUILD_TRACE)
  message(FATAL_ERROR "Trace fault-injection tests require MGMEM_IMAGE_BUILD_TRACE")
endif()
if(MGMEM_IMAGE_BUILD_TRACE)
  add_executable(mgmem_image_trace paired.cpp)
  target_compile_features(mgmem_image_trace PRIVATE cxx_std_17)
  target_compile_definitions(mgmem_image_trace PRIVATE MGMEM_IMAGE_TRACE)
  target_link_libraries(mgmem_image_trace PRIVATE MeguvizMemory::mgmem Threads::Threads)
  if(MGMEM_IMAGE_BUILD_TRACE_TESTS)
    add_executable(mgmem_image_trace_fault paired.cpp)
    target_compile_features(mgmem_image_trace_fault PRIVATE cxx_std_17)
    target_compile_definitions(mgmem_image_trace_fault PRIVATE MGMEM_IMAGE_TRACE MGMEM_IMAGE_TRACE_TESTING)
    target_link_libraries(mgmem_image_trace_fault PRIVATE MeguvizMemory::mgmem Threads::Threads)
    add_executable(mgmem_image_trace_sha256 test_trace_sha256.cpp)
    target_compile_features(mgmem_image_trace_sha256 PRIVATE cxx_std_17)
  endif()
endif()
target_compile_features(mgmem_image_pressure_stability PRIVATE cxx_std_17)
target_link_libraries(mgmem_image_pressure_stability PRIVATE MeguvizMemory::pressure Threads::Threads)
target_compile_features(mgmem_image PRIVATE cxx_std_17)
target_link_libraries(mgmem_image PRIVATE MeguvizMemory::mgmem Threads::Threads)
target_compile_features(mgmem_image_paired PRIVATE cxx_std_17)
target_link_libraries(mgmem_image_paired PRIVATE MeguvizMemory::mgmem Threads::Threads)
target_compile_features(mgmem_image_perf PRIVATE cxx_std_17)
target_link_libraries(mgmem_image_perf PRIVATE MeguvizMemory::mgmem Threads::Threads)
if(MSVC)
  target_compile_options(mgmem_image PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(mgmem_image_paired PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(mgmem_image_perf PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(mgmem_image_pressure_stability PRIVATE /W4 /permissive- /utf-8)
  if(MGMEM_IMAGE_BUILD_TRACE)
    target_compile_options(mgmem_image_trace PRIVATE /W4 /permissive- /utf-8)
  endif()
  if(MGMEM_IMAGE_BUILD_TRACE_TESTS)
    target_compile_options(mgmem_image_trace_fault PRIVATE /W4 /permissive- /utf-8)
    target_compile_options(mgmem_image_trace_sha256 PRIVATE /W4 /permissive- /utf-8)
  endif()
else()
  target_compile_options(mgmem_image PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(mgmem_image_paired PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(mgmem_image_perf PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(mgmem_image_pressure_stability PRIVATE -Wall -Wextra -Wpedantic)
  if(MGMEM_IMAGE_BUILD_TRACE)
    target_compile_options(mgmem_image_trace PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(MGMEM_IMAGE_BUILD_TRACE_TESTS)
    target_compile_options(mgmem_image_trace_fault PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(mgmem_image_trace_sha256 PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
