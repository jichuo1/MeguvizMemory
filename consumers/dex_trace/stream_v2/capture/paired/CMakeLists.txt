cmake_minimum_required(VERSION 3.25)
project(MeguvizDexConsumerReaderStreamV2 LANGUAGES C CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
get_filename_component(CAPTURE_DIR "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(DEX_TRACE_STAGE "" CACHE PATH "Verified private Dex source preparation")
set(PAIRED_TRACE_STAGE "" CACHE PATH "Verified private V2 paired preparation")
set(DEX_TRACE_MODE "" CACHE STRING "baseline or traced")
if(NOT DEX_TRACE_MODE MATCHES "^(baseline|traced)$" OR
   NOT EXISTS "${DEX_TRACE_STAGE}/manifest.json" OR
   NOT EXISTS "${PAIRED_TRACE_STAGE}/manifest.json" OR
   NOT EXISTS "${PAIRED_TRACE_STAGE}/paired_stream.cpp")
  message(FATAL_ERROR "Supply both verified stages and an explicit baseline/traced mode")
endif()
find_package(MeguvizMemory CONFIG REQUIRED)
set(DMG_BUILD_TESTS OFF CACHE BOOL "No external parser tests" FORCE)
set(DMG_WITH_SQLITE OFF CACHE BOOL "No optional SQL bridge needed" FORCE)
add_subdirectory("${DEX_TRACE_STAGE}/${DEX_TRACE_MODE}" "${CMAKE_CURRENT_BINARY_DIR}/dexmeguviz" EXCLUDE_FROM_ALL)
add_executable(mgmem_dex_reader_stream_v2 "${PAIRED_TRACE_STAGE}/paired_stream.cpp" "${CAPTURE_DIR}/stream_trace.cpp")
target_compile_features(mgmem_dex_reader_stream_v2 PRIVATE cxx_std_20)
target_include_directories(mgmem_dex_reader_stream_v2 PRIVATE "${CAPTURE_DIR}" "${ROOT}/consumers/dex_trace" "${ROOT}/consumers/image")
target_link_libraries(mgmem_dex_reader_stream_v2 PRIVATE MeguvizMemory::mgmem dexmeguviz_core ${CMAKE_DL_LIBS})
if(DEX_TRACE_MODE STREQUAL "traced")
  target_compile_definitions(mgmem_dex_reader_stream_v2 PRIVATE MGMEM_DEX_CONSUMER_TRACE_STREAM_V2=1)
  target_include_directories(dexmeguviz_core PRIVATE "${ROOT}/consumers/dex_trace")
endif()
if(MSVC)
  target_compile_options(mgmem_dex_reader_stream_v2 PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(mgmem_dex_reader_stream_v2 PRIVATE -Wall -Wextra -Wpedantic)
endif()
