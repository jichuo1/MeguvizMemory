cmake_minimum_required(VERSION 3.25)
project(MeguvizDexMetadataTrace LANGUAGES CXX)
set(DEX_TRACE_COPY "" CACHE PATH "Private prepare.py output")
if(NOT EXISTS "${DEX_TRACE_COPY}/manifest.json")
    message(FATAL_ERROR "DEX_TRACE_COPY must point at a prepared private snapshot")
endif()
foreach(mode IN ITEMS baseline instrumented)
    add_executable(dex_metadata_${mode} harness.cpp "${DEX_TRACE_COPY}/${mode}/reader.cpp")
    target_compile_features(dex_metadata_${mode} PRIVATE cxx_std_20)
    target_include_directories(dex_metadata_${mode} PRIVATE "${DEX_TRACE_COPY}/${mode}" "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_LIST_DIR}/../image")
    if(mode STREQUAL "instrumented")
        target_compile_definitions(dex_metadata_${mode} PRIVATE MGMEM_DEX_METADATA_TRACE=1)
    endif()
endforeach()
