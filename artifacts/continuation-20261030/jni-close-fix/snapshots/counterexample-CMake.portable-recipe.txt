cmake_minimum_required(VERSION 3.25)
project(JniLockReview LANGUAGES CXX)
find_package(MeguvizMemory CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(JDK_INCLUDE "$ENV{JAVA_HOME}/include")
foreach(mode old new)
  add_executable(jni_lock_${mode} fake_jni_test.cpp)
  target_compile_features(jni_lock_${mode} PRIVATE cxx_std_17)
  target_compile_options(jni_lock_${mode} PRIVATE /EHsc /W4 /utf-8)
  target_include_directories(jni_lock_${mode} PRIVATE "${JDK_INCLUDE}" "${JDK_INCLUDE}/win32")
  target_compile_definitions(jni_lock_${mode} PRIVATE JNI_REVIEW_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/pressure_jni.host-${mode}.cpp")
  target_link_libraries(jni_lock_${mode} PRIVATE MeguvizMemory::pressure Threads::Threads)
endforeach()
