root=Path(__file__).resolve().parents[2]
parser=argparse.ArgumentParser(description=__doc__)

parser.add_argument('--abi',choices=['arm64-v8a','x86_64'],required=True)
parser.add_argument('--label',required=True)
parser.add_argument('--expected-source-sha256',required=True)
args=parser.parse_args()

source=args.source.resolve();target=root/'build/android-close-shareable/delivery-v2'/('jni-'+args.label)/args.abi/'libmgmem_pressure_jni.so'

receipt=root/'artifacts/android-close-investigation-20261030/shareable-sdk-v2'/('jni-'+args.label+'-'+args.abi+'.json')
if receipt.exists():raise RuntimeError('refuse_to_overwrite_JNI_receipt')
value={'state':'PASS_STRIP_AND_ELF_CONTRACT_DEVICE_NOT_EXECUTED','ABI':args.abi,'JNI_source_sha256':args.expected_source_sha256,'transformation':result,'helper_sha256':sha(Path(__file__)),'executed_binary_contract_helper_sha256':sha(root/'build/android-close-shareable/prepare_shareable.py'),'prior_run_byte_identity':'DISTINCT_IF_SHA_DIFFERS_NEEDS_NEW_ANDROID_VALIDATION','production_input_unchanged':True}
receipt.write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n',encoding='utf-8',newline='\n')
print(json.dumps({'state':value['state'],'receipt':receipt.relative_to(root).as_posix(),'candidate':target.relative_to(root).as_posix(),'sha256':sha(target)}))
