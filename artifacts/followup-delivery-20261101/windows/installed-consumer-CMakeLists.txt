cmake_minimum_required(VERSION 3.25)
project(InstalledMemoryVerification LANGUAGES C CXX)
find_package(MeguvizMemory CONFIG REQUIRED)
find_package(JNI REQUIRED)
find_package(Threads REQUIRED)
add_executable(checked_cpp "${SOURCE_ROOT}/tests/cpp_checked_import_tests.cpp")
target_compile_features(checked_cpp PRIVATE cxx_std_17)
target_link_libraries(checked_cpp PRIVATE MeguvizMemory::mgmem Threads::Threads)
add_executable(legacy_cpp "${SOURCE_ROOT}/tests/cpp_consumer.cpp")
target_compile_features(legacy_cpp PRIVATE cxx_std_17)
target_link_libraries(legacy_cpp PRIVATE MeguvizMemory::mgmem Threads::Threads)
add_executable(checked_c "${SOURCE_ROOT}/tests/checked_import_c.c")
target_link_libraries(checked_c PRIVATE MeguvizMemory::mgmem)
add_library(installed_jni SHARED IMPORTED)
set_target_properties(installed_jni PROPERTIES IMPORTED_LOCATION "${JNI_PREFIX}/bin/mgmem_pressure_jni.dll" IMPORTED_IMPLIB "${JNI_PREFIX}/lib/mgmem_pressure_jni.lib")
add_library(mgmem_fixture SHARED "${SOURCE_ROOT}/tests/borrow_jni_java_fixture.cpp")
target_compile_features(mgmem_fixture PRIVATE cxx_std_17)
target_compile_definitions(mgmem_fixture PRIVATE MGMEM_PRESSURE_SHARED)
target_include_directories(mgmem_fixture PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(mgmem_fixture PRIVATE MeguvizMemory::mgmem installed_jni Threads::Threads)
enable_testing()
foreach(c normal owner-failure borrow-failure destructor-failure borrow-destructor move-assignment rollback-failure concurrent)
 add_test(NAME checked_${c} COMMAND checked_cpp ${c})
endforeach()
add_test(NAME checked_c COMMAND checked_c)
add_test(NAME legacy_cpp COMMAND legacy_cpp)
