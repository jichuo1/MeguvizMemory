cmake_minimum_required(VERSION 3.25)
project(MeguvizMemoryBorrowProbe LANGUAGES CXX)
if(NOT ANDROID)
  message(FATAL_ERROR "This fixture is built only with the Android NDK")
endif()
if(NOT IS_ABSOLUTE "${MGMEM_ANDROID_JNI_LIBRARY}" OR NOT EXISTS "${MGMEM_ANDROID_JNI_LIBRARY}")
  message(FATAL_ERROR "MGMEM_ANDROID_JNI_LIBRARY must name the test input AAR JNI library")
endif()
if(NOT IS_ABSOLUTE "${MGMEM_ANDROID_SDK_DIR}" OR NOT EXISTS "${MGMEM_ANDROID_SDK_DIR}/lib/cmake/MeguvizMemory/MeguvizMemoryConfig.cmake")
  message(FATAL_ERROR "MGMEM_ANDROID_SDK_DIR must name the installed Android Memory SDK")
endif()
find_package(MeguvizMemory CONFIG REQUIRED
  PATHS "${MGMEM_ANDROID_SDK_DIR}/lib/cmake/MeguvizMemory" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
get_target_property(memory_kind MeguvizMemory::mgmem TYPE)
if(NOT memory_kind STREQUAL "SHARED_LIBRARY")
  message(FATAL_ERROR "Fixture requires one shared Memory runtime")
endif()
add_library(mgmem_android_jni SHARED IMPORTED)
set_target_properties(mgmem_android_jni PROPERTIES IMPORTED_LOCATION "${MGMEM_ANDROID_JNI_LIBRARY}")
add_library(mgmem_borrow_fixture SHARED borrow_fixture.cpp)
target_compile_features(mgmem_borrow_fixture PRIVATE cxx_std_17)
target_compile_options(mgmem_borrow_fixture PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(mgmem_borrow_fixture PRIVATE mgmem_android_jni MeguvizMemory::mgmem)
target_link_options(mgmem_borrow_fixture PRIVATE "-Wl,-z,max-page-size=16384")
set_target_properties(mgmem_borrow_fixture PROPERTIES CXX_EXTENSIONS OFF CXX_VISIBILITY_PRESET hidden)
