cmake_minimum_required(VERSION 3.25)
project(MeguvizMemoryCppCheckedDevelopment LANGUAGES CXX)
find_package(Threads REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)

add_executable(cpp_checked_import "${ROOT}/src/mgmem.cpp" "${ROOT}/tests/cpp_checked_import_tests.cpp")
target_compile_features(cpp_checked_import PRIVATE cxx_std_17)
target_include_directories(cpp_checked_import PRIVATE "${ROOT}/include")
target_link_libraries(cpp_checked_import PRIVATE Threads::Threads)

add_executable(cpp_checked_fake "${ROOT}/tests/cpp_checked_fake_tests.cpp")
target_compile_features(cpp_checked_fake PRIVATE cxx_std_17)
target_include_directories(cpp_checked_fake PRIVATE "${ROOT}/include")
target_link_libraries(cpp_checked_fake PRIVATE Threads::Threads)

add_executable(cpp_legacy_consumer "${ROOT}/src/mgmem.cpp" "${ROOT}/tests/cpp_consumer.cpp")
target_compile_features(cpp_legacy_consumer PRIVATE cxx_std_17)
target_include_directories(cpp_legacy_consumer PRIVATE "${ROOT}/include")
target_link_libraries(cpp_legacy_consumer PRIVATE Threads::Threads)

if(MSVC)
  target_compile_options(cpp_checked_import PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(cpp_checked_fake PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(cpp_legacy_consumer PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(cpp_checked_import PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(cpp_checked_fake PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(cpp_legacy_consumer PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
foreach(case normal owner-failure borrow-failure destructor-failure borrow-destructor move-assignment rollback-failure concurrent)
  add_test(NAME cpp_checked_${case} COMMAND cpp_checked_import ${case})
  set_tests_properties(cpp_checked_${case} PROPERTIES TIMEOUT 20)
endforeach()
add_test(NAME cpp_checked_fake_unconsumed COMMAND cpp_checked_fake)
set_tests_properties(cpp_checked_fake_unconsumed PROPERTIES TIMEOUT 20)
add_test(NAME cpp_legacy_consumer COMMAND cpp_legacy_consumer)
set_tests_properties(cpp_legacy_consumer PROPERTIES TIMEOUT 30)
