cmake_minimum_required(VERSION 3.25)
project(MeguvizMemoryAndroidPressure LANGUAGES CXX)
if(NOT ANDROID)
  message(FATAL_ERROR "Use the Android NDK toolchain for the JNI composition")
endif()
find_package(MeguvizMemory CONFIG REQUIRED)
find_package(Threads REQUIRED)
include(GNUInstallDirs)
if(NOT ANDROID_ABI MATCHES "^[A-Za-z0-9_-]+$")
  message(FATAL_ERROR "A valid Android ABI is required for the JNI installation")
endif()
get_target_property(memory_kind MeguvizMemory::mgmem TYPE)
if(NOT memory_kind STREQUAL "SHARED_LIBRARY")
  message(FATAL_ERROR "This standalone JNI composition requires the shared Memory SDK; do not embed a second runtime")
endif()
add_library(mgmem_pressure_jni SHARED native/pressure_jni.cpp native/borrow_jni.cpp native/workspace_jni.cpp)
target_compile_features(mgmem_pressure_jni PRIVATE cxx_std_17)
target_link_libraries(mgmem_pressure_jni PUBLIC MeguvizMemory::mgmem PRIVATE MeguvizMemory::pressure Threads::Threads)
target_compile_options(mgmem_pressure_jni PRIVATE -Wall -Wextra -Wpedantic)
target_link_options(mgmem_pressure_jni PRIVATE "-Wl,-z,max-page-size=16384")
set_target_properties(mgmem_pressure_jni PROPERTIES EXPORT_NAME android_jni CXX_EXTENSIONS OFF CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
install(TARGETS mgmem_pressure_jni EXPORT MeguvizMemoryAndroidJniTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(EXPORT MeguvizMemoryAndroidJniTargets NAMESPACE MeguvizMemory::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizMemory)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/MeguvizMemoryAndroidJniABI.cmake"
  "set(MeguvizMemory_ANDROID_JNI_ABI \"${ANDROID_ABI}\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/MeguvizMemoryAndroidJniABI.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizMemory)
